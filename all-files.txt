const fs = require("fs");
const path = require("path");

const SKIP_DIRS = ["node_modules", ".next", ".git", "dist", "build"];
const SKIP_FILES = ["package-lock.json", "yarn.lock", "pnpm-lock.yaml"];
const EXTENSIONS = [
  ".tsx",
  ".ts",
  ".js",
  ".jsx",
  ".css",
  ".scss",
  ".json",
  ".md",
];

let output = "";
let count = 0;

function walk(dir) {
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (!SKIP_DIRS.includes(entry.name)) walk(full);
    } else if (
      !SKIP_FILES.includes(entry.name) &&
      EXTENSIONS.includes(path.extname(entry.name))
    ) {
      output += `\n\n================ FILE: ${path.relative(".", full)} ================\n\n`;
      output += fs.readFileSync(full, "utf8");
      count++;
    }
  }
}

walk(".");
fs.writeFileSync("all-files.txt", output);
console.log(`✅ Done! Combined ${count} files into all-files.txt`);

const CACHE = "pos-v1"; // bump this version when you deploy UI updates
const CORE = [
  "/",
  "/pos",
  "/products",
  "/customers",
  "/reports",
  "/settings",
  "/activation",
];
self.addEventListener("install", (e) => {
  e.waitUntil(
    Promise.allSettled(
      CORE.map((u) => caches.open(CACHE).then((c) => c.add(u))),
    ),
  );
  self.skipWaiting();
});
self.addEventListener("activate", (e) => {
  e.waitUntil(
    caches
      .keys()
      .then((ks) =>
        Promise.all(ks.filter((k) => k !== CACHE).map((k) => caches.delete(k))),
      )
      .then(() => self.clients.claim()),
  );
});
self.addEventListener("fetch", (e) => {
  const url = new URL(e.request.url);
  if (url.origin !== location.origin || e.request.method !== "GET") return;
  if (url.pathname.startsWith("/api/") || url.pathname.startsWith("/admin"))
    return;
  e.respondWith(
    fetch(e.request)
      .then((res) => {
        const copy = res.clone();
        caches.open(CACHE).then((c) => c.put(e.request, copy));
        return res;
      })
      .catch(() => caches.match(e.request).then((h) => h || caches.match("/"))),
  );
});

  const { t } = useT();
  const [form, setForm] = useState<any>(null);
  const [hist, setHist] = useState<Customer | null>(null);
  const list = useLiveQuery(() => db.customers.orderBy("name").toArray(), []);
  const histSales = useLiveQuery(
    () =>
      hist
        ? db.sales.where("customerId").equals(hist.id!).reverse().sortBy("ts")
        : [],
    [hist],
  );

  async function save() {
    const f = form;
    try {
      if (f.id)
        await db.customers.update(f.id, {
          name: f.name.trim(),
          phone: f.phone.trim(),
          address: f.address?.trim(),
        });
      else
        await db.customers.add({
          phone: f.phone.trim(),
          name: f.name.trim() || f.phone.trim(),
          address: f.address?.trim(),
          points: 0,
          balance: 0,
        });
      toast.success(t("saved"));
      setForm(null);
    } catch {
      toast.error(t("phone") + " — duplicate");
    }
  }
  async function adjust(c: Customer, d: number) {
    await db.customers.update(c.id!, { balance: Math.max(0, c.balance + d) });
  }

  return (
    <div className="space-y-3">
      <div className="flex justify-end">

                variant="outline"
                onClick={() => adjust(c, -c.balance)}
              >
                Clear
              </Button>

"use client";
import { Card, CardContent } from "@/components/ui/card";
import { db, getSetting } from "@/lib/db";
import { dayStart, money } from "@/lib/format";
import { useT } from "@/lib/i18n";
import { useLiveQuery } from "dexie-react-hooks";
import {
  AlertTriangle,
  PackageX,
  ShoppingBag,
  TrendingUp,
  Wallet,
} from "lucide-react";
import Link from "next/link";

export default function Dashboard() {
  const { t } = useT();
  const agg = useLiveQuery(async () => {
    const sales = await db.sales
      .where("ts")
      .between(dayStart(), Date.now())
      .toArray();
    const paid = sales.filter((s) => s.status === "paid");
    const revenue = paid.reduce((a, s) => a + s.total, 0);
    const low = (await db.variants.toArray()).filter(
      (v) => v.stock <= v.reorderLevel,
    );
    const recent = await db.sales.orderBy("ts").reverse().limit(6).toArray();
    const lastBackup = (await getSetting("lastBackupAt")) as number | undefined;
    return {
      revenue,
      count: paid.length,
      low,
      recent,
      backupStale: !lastBackup || Date.now() - lastBackup > 7 * 86_400_000,
    };
  }, []);
  if (!agg) return null;
  const K = [
    {
      icon: TrendingUp,
      label: t("revenue"),
      v: money(agg.revenue),
      cls: "text-emerald-600",
    },
    {
      icon: ShoppingBag,
      label: t("transactions"),
      v: String(agg.count),
      cls: "",
    },
    {
      icon: Wallet,
      label: t("avgBasket"),
      v: money(agg.count ? agg.revenue / agg.count : 0),
      cls: "",
    },
    {
      icon: PackageX,
      label: t("lowStock"),
      v: String(agg.low.length),
      cls: "text-red-600",
    },
  ];
  return (
    <div className="space-y-4">
      <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
        {K.map((k) => (
          <Card key={k.label}>
            <CardContent className="p-4">
              <div className="flex items-center justify-between text-xs text-muted-foreground">
                <span>{k.label}</span>
                <k.icon className="size-4" />
              </div>
              <p className={`num mt-1 text-xl font-semibold ${k.cls}`}>{k.v}</p>
            </CardContent>
          </Card>
        ))}
      </div>
      {agg.backupStale && (
        <Link
          href="/settings"
          className="flex items-center gap-2 rounded-md border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-800"
        >
          <AlertTriangle className="size-4 shrink-0" />
          {t("backupReminder")}
        </Link>
      )}
      <Card>
        <CardContent className="p-4">
          <h2 className="mb-2 text-sm font-semibold">{t("recentSales")}</h2>
          {agg.recent.length === 0 && (
            <p className="text-sm text-muted-foreground">{t("noData")}</p>
          )}
          <div className="divide-y">
            {agg.recent.map((s) => (
              <div
                key={s.id}
                className="flex items-center justify-between py-2 text-sm"
              >
                <span className="text-muted-foreground">
                  #{s.id} · {new Date(s.ts).toLocaleString("en-GB")}

                <span
                  className={`num font-semibold ${s.status === "paid" ? "text-emerald-600" : "text-muted-foreground"}`}
                >
                  {money(s.total)} · {s.status}
                </span>
              </div>
            ))}
          </div>
        </CardContent>
      </Card>
    </div>
  );
}

  const totals = useMemo(() => {
    const subtotal = cart.reduce((a, l) => a + l.unitPrice * l.qty, 0);
    const discount = cart.reduce((a, l) => a + l.discount, 0);
    const tax = Math.round((subtotal - discount) * taxRate) / 100;
    return {
      subtotal,
      discount,
      tax,
      total: Math.round(subtotal - discount + tax),
    };
  }, [cart, taxRate]);

  function add(v: any) {
    const inCart = cart.find((l) => l.variantId === v.id)?.qty ?? 0;
    if (v.stock - inCart < 1) return toast.error(t("outOfStock"));
    setCart((c) => {
      const i = c.findIndex((l) => l.variantId === v.id);
      if (i >= 0) {
        const n = [...c];
        n[i] = { ...n[i], qty: n[i].qty + 1 };
        return n;
      }
      return [
        ...c,
        {
          variantId: v.id!,
          sku: v.sku,
          name: v.name,
          nameUr: v.nameUr,
          size: v.size,
          color: v.color,
          qty: 1,
          unitPrice: v.price,
          cost: v.costPrice,
          discount: 0,
        },
      ];
    });
  }
  const bump = (id: number, d: number) =>
    setCart((c) =>
      c
        .map((l) => (l.variantId === id ? { ...l, qty: l.qty + d } : l))
        .filter((l) => l.qty > 0),
    );

  async function makeSale(status: "paid" | "held"): Promise<Sale | null> {
    const r = await evaluateLicense();
    if (!canSell(r.state)) {
      toast.error(t("subExpired"));
      return null;
    }
    if (cart.length === 0) return null;
    let customerId: number | undefined;
    if (phone.trim()) {
      const ex = await db.customers.where("phone").equals(phone.trim()).first();
      customerId =
        ex?.id ??
        (await db.customers.add({
          phone: phone.trim(),
          name: phone.trim(),
          points: 0,
          balance: 0,
        }));
    }
    const receiptLang = ((await getSetting("receiptLang")) ?? "en") as
      | "en"
      | "ur"
      | "both";
    const payAmount =
      method === "cash" ? Number(amountPaid || totals.total) : totals.total;
    if (status === "paid" && method === "cash" && payAmount < totals.total) {
      toast.error(t("insufficient"));
      return null;
    }
    const sale: Sale = {
      ts: Date.now(),
      customerId,
      items: cart.map(({ ...i }) => ({ ...i })),
      ...totals,
      payments: status === "held" ? [] : [{ method, amount: payAmount }],
      status,
      receiptLang,
    };
    const id = await db.transaction(
      "rw",
      db.sales,
      db.variants,
      db.moves,
      db.customers,
      async () => {
        const sid = await db.sales.add(sale);
        if (status === "paid") {
          for (const l of cart) {
            const v = (await db.variants.get(l.variantId))!;
            await db.variants.update(l.variantId, { stock: v.stock - l.qty });
            await db.moves.add({
              ts: Date.now(),
              variantId: l.variantId,
              qty: -l.qty,
              type: "sale",
              ref: `S${sid}`,
            });
          }
          if (method === "credit" && customerId) {
            const c = (await db.customers.get(customerId))!;
            await db.customers.update(customerId, {
              balance: c.balance + totals.total,
            });
          }
        }
        return sid;
      },
    );
    const done = { ...sale, id };
    setCart([]);
    setAmountPaid("");
    setPhone("");
    if (status === "paid") setLastSale(done);
    else toast.success(t("saved"));
    return done;
  }

  const held = useLiveQuery(
    () => db.sales.where("status").equals("held").reverse().sortBy("ts"),
    [],
  );
  async function recall(id: number) {
    const s = await db.sales.get(id);
    if (!s) return;
    setCart(s.items.map((i) => ({ ...i })));
    await db.sales.delete(id);
  }

  if (!sellable)

      <div className="grid place-items-center rounded-lg border border-dashed py-24 text-center">

import { Button } from "@/components/ui/button";

  const [from, setFrom] = useState(ymd(new Date()));
  const [to, setTo] = useState(ymd(new Date()));
  const [range, setRange] = useState({ from: from, to: to });

  const data = useLiveQuery(async () => {
    const start = new Date(range.from + "T00:00:00").getTime();
    const end = new Date(range.to + "T23:59:59").getTime();
    const sales = (
      await db.sales.where("ts").between(start, end).toArray()
    ).filter((s) => s.status === "paid");
    const revenue = sales.reduce((a, s) => a + s.total, 0);
    let profit = 0;
    const bySku = new Map<
      string,
      { name: string; qty: number; revenue: number; profit: number }
    >();
    const mix = new Map<string, number>();
    for (const s of sales) {
      for (const i of s.items) {
        profit += i.qty * i.unitPrice - i.discount - i.qty * i.cost;
        const r = bySku.get(i.sku) ?? {
          name: i.name,
          qty: 0,
          revenue: 0,
          profit: 0,
        };
        r.qty += i.qty;
        r.revenue += i.qty * i.unitPrice - i.discount;
        r.profit += i.qty * i.unitPrice - i.discount - i.qty * i.cost;
        bySku.set(i.sku, r);
      }
      for (const p of s.payments)
        mix.set(p.method, (mix.get(p.method) ?? 0) + p.amount);
    }
    const low = (await db.variants.toArray()).filter(
      (v) => v.stock <= v.reorderLevel,
    );
    return {
      revenue,
      profit,
      count: sales.length,
      rows: [...bySku.values()].sort((a, b) => b.revenue - a.revenue),
      mix: [...mix.entries()],
      low,
    };
  }, [range]);

  const cur = "Rs";
  function csv() {
    const esc = (v: any) => `"${String(v).replace(/"/g, '""')}"`;
    const lines = [
      ["SKU", "Name", "Qty", "Revenue", "Profit"],

import {

  deviceSecret,
  evaluateLicense,
  type LicensePayload,
} from "@/lib/license";
import { RefreshCw, Store } from "lucide-react";
import Link from "next/link";
import { useEffect, useState } from "react";

const ERR: Record<string, string> = {
  INVALID_CODE: "invalidCode",
  DEVICE_MISMATCH: "deviceMismatch",
  REVOKED: "revoked",
  TOO_MANY: "tooMany",
  UNKNOWN_SHOP: "unknownShop",
};

export default function Activation() {
  const { t } = useT();
  const [snap, setSnap] = useState<{
    state: string;
    daysLeft: number;
    lic: LicensePayload | null;
  } | null>(null);
  const [cid, setCid] = useState("");
  const [code, setCode] = useState("");
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState("");

  useEffect(() => {
    evaluateLicense().then(setSnap);
  }, []);
  const state = snap?.state ?? "none";
  const lic = snap?.lic ?? null;

  async function updateOnline() {
    setBusy(true);
    setMsg("");
    try {
      const ds = await deviceSecret();
      const fp = await fingerprint();
      const id = (lic?.cid ?? cid).trim().toUpperCase();
      if (!id) {
        setMsg(t("enterShopId"));
        return;
      }
      const r = await fetch(
        `/api/license?cid=${encodeURIComponent(id)}&fp=${fp}&ds=${encodeURIComponent(ds)}`,
      );
      const j = await r.json();
      if (!r.ok) {
        setMsg(t(ERR[j.error] ?? "needInternet"));
        return;
      }
      if (!clockIsSane(j.serverTime)) {
        setMsg(t("clockIssue"));
        return;
      }
      const saved = await activateWithCode(j.code, ds);
      setMsg(`${t("validUntil")} ${saved.exp}`);
      evaluateLicense().then(setSnap);
    } catch {
      setMsg(t("needInternet"));
    } finally {
      setBusy(false);
    }
  }
  async function paste() {
    setMsg("");
    try {
      const ds = await deviceSecret();
      await activateWithCode(code, ds);
      location.href = "/";
    } catch (e: any) {
      setMsg(t(ERR[e?.message] ?? "invalidCode"));
    }
  }

  const tone =
    state === "locked" || state === "tampered"
      ? "text-red-600"
      : state === "grace"
        ? "text-red-500"
        : state === "warning"
          ? "text-amber-600"
          : "text-emerald-600";
  const statusText =
    state === "none"
      ? t("enterShopId")
      : state === "locked" || state === "tampered"
        ? t("subExpired")
        : state === "grace"
          ? t("graceNote")
          : state === "warning"
            ? `${snap!.daysLeft} ${t("daysLeft")} — ${t("payReminder")}`
            : `${t("validUntil")} ${lic?.exp}`;

  return (
    <main className="grid min-h-screen place-items-center bg-muted/40 p-4">
      <div className="w-full max-w-md space-y-4 rounded-xl border bg-background p-8 shadow-sm">
        <div className="flex items-center justify-center gap-2">
          <Store className="size-6" />
          <h1 className="text-lg font-semibold">{t("appName")}</h1>
        </div>
        {lic?.branding?.shopName ? (
          <p className="text-center text-sm text-muted-foreground">
            {String(lic.branding.shopName)}
          </p>
        ) : null}
        <p className={`text-center text-sm font-medium ${tone}`}>
          {statusText}
        </p>
        {!lic?.cid && (
          <Input
            value={cid}
            onChange={(e) => setCid(e.target.value)}
            placeholder={t("shopId")}
            className="text-center uppercase"
          />
        )}
        <Button className="w-full" onClick={updateOnline} disabled={busy}>
          {busy ? <RefreshCw className="mr-2 size-4 animate-spin" /> : null}
          {t("updateSubscription")}
        </Button>
        <p className="text-center text-xs text-muted-foreground">
          {t("needsInternetOnce")}
        </p>
        <div className="space-y-2 border-t pt-3">
          <p className="text-xs text-muted-foreground">{t("orPasteCode")}</p>
          <Textarea
            value={code}
            onChange={(e) => setCode(e.target.value)}
            rows={3}
            className="text-xs"
          />
          <Button
            variant="outline"
            className="w-full"
            onClick={paste}
            disabled={!code.trim()}
          >
            Activate
          </Button>
        </div>
        {msg && (
          <p className="text-center text-sm font-medium text-blue-600">{msg}</p>
        )}
        {snap && ["active", "warning", "grace"].includes(state) && (
          <Link
            href="/"
            className="block text-center text-sm font-semibold text-emerald-600 hover:underline"
          >
            {t("openApp")} →
          </Link>
        )}
      </div>
    </main>
  );
}

    await jwtVerify(
      tk!,
      new TextEncoder().encode(process.env.ADMIN_JWT_SECRET!),
    );
  } catch {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });

    return NextResponse.json({ error: "UNKNOWN_SHOP" }, { status: 404 });
  if (!rec.ds)
    return NextResponse.json({ error: "NEVER_ACTIVATED" }, { status: 400 });
  const payload = {
    cid: rec.cid,
    fp: rec.fp ?? "",
    ds: rec.ds,
    iat: Date.now(),
    plan: rec.plan ?? "monthly",
    status: "active" as const,
    exp: rec.expiry,
    features: rec.features ?? {},

    JSON.stringify({ d: payload, s: Buffer.from(sig).toString("hex") }),

    .slice(0, max)
    .trim();

export async function POST(req: NextRequest) {
  if (!(await auth(req)))
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  const b = await req.json();
  const cid = clean(b.cid, 24).toUpperCase();
  if (!/^[A-Z0-9_-]{2,24}$/.test(cid))
    return NextResponse.json({ error: "BAD_CID" }, { status: 400 });
  const prev = (await kvGet<any>(`cust:${cid}`)) ?? {};
  let ds = prev.ds;
  if (b.ds === null) ds = null;
  else if (typeof b.ds === "string" && b.ds) ds = b.ds;
  const rec = {
    cid,
    status: b.status === "revoked" ? "revoked" : "active",
    plan: ["monthly", "quarterly", "yearly"].includes(b.plan)
      ? b.plan
      : "monthly",
    expiry: /^\d{4}-\d{2}-\d{2}$/.test(b.expiry ?? "")
      ? b.expiry
      : "1970-01-01",
    ds,
    fp: prev.fp ?? "",
    features: Object.fromEntries(
      Object.entries(b.features ?? {}).map(([k, v]) => [k, v === true]),
    ),
    branding: {
      shopName: clean(b.branding?.shopName, 60),
      shopNameUr: clean(b.branding?.shopNameUr, 60),
      address: clean(b.branding?.address, 200),
      phone: clean(b.branding?.phone, 30),
      footerEn: clean(b.branding?.footerEn, 200),
      footerUr: clean(b.branding?.footerUr, 200),
      currency: clean(b.branding?.currency, 8) || "Rs",
      taxRate: Math.max(0, Math.min(50, Number(b.branding?.taxRate) || 0)),
      taxName: clean(b.branding?.taxName, 12) || "Tax",
    },
    updatedAt: Date.now(),
  };
  await kvSet(`cust:${cid}`, rec);
  const idx = (await kvGet<string[]>("cust_index")) ?? [];
  if (!idx.includes(cid)) await kvSet("cust_index", [...idx, cid]);
  return NextResponse.json({ ok: true });
}

import { NextRequest, NextResponse } from "next/server";
export async function POST(req: NextRequest) {
  const { password } = await req.json();
  if (password !== process.env.ADMIN_PASSWORD)
    return NextResponse.json({ error: "wrong" }, { status: 401 });
  const token = await new SignJWT({ r: "owner" })
    .setProtectedHeader({ alg: "HS256" })
    .setExpirationTime("7d")
    .sign(new TextEncoder().encode(process.env.ADMIN_JWT_SECRET!));
  const res = NextResponse.json({ ok: true });
  res.cookies.set("admin_sess", token, {
    httpOnly: true,
    secure: true,
    sameSite: "lax",
    maxAge: 604800,
    path: "/",
  });
  return res;
}

// src/app/api/license/route.ts
import { kvGet, kvIncr, kvSet } from "@/lib/server/kv";
import { NextRequest, NextResponse } from "next/server";
import nacl from "tweetnacl";

const hexToBytes = (h: string) =>
  new Uint8Array(h.match(/../g)!.map((x) => parseInt(x, 16)));

export async function GET(req: NextRequest) {
  const ip = (req.headers.get("x-forwarded-for") ?? "x").split(",")[0].trim();
  if ((await kvIncr(`rl:${ip}`)) > 20)
    return NextResponse.json({ error: "TOO_MANY" }, { status: 429 });

  const q = req.nextUrl.searchParams;
  const cid = (q.get("cid") ?? "").trim().toUpperCase();
  const fp = (q.get("fp") ?? "").trim();
  const ds = (q.get("ds") ?? "").trim();
  if (!cid || !ds)
    return NextResponse.json({ error: "MISSING_PARAMS" }, { status: 400 });

  const rec = await kvGet<any>(`cust:${cid}`);
  if (!rec)
    return NextResponse.json({ error: "UNKNOWN_SHOP" }, { status: 404 });
  if (rec.status === "revoked")
    return NextResponse.json({ error: "REVOKED" }, { status: 403 });
  if (rec.ds && rec.ds !== ds)
    return NextResponse.json({ error: "DEVICE_MISMATCH" }, { status: 403 });
  if (!rec.ds) {
    rec.ds = ds;
    rec.fp = fp;
    await kvSet(`cust:${cid}`, rec);
  }

  const payload = {
    cid,
    fp,
    ds,
    iat: Date.now(),
    plan: rec.plan ?? "monthly",
    status: "active" as const,
    exp: rec.expiry,
    features: rec.features ?? {},
    branding: rec.branding ?? {},
  };
  const sig = nacl.sign.detached(
    new TextEncoder().encode(JSON.stringify(payload)),
    hexToBytes(process.env.LICENSE_PRIVATE_KEY!),
  );
  const code = Buffer.from(
    JSON.stringify({ d: payload, s: Buffer.from(sig).toString("hex") }),
  ).toString("base64");
  return NextResponse.json({ code, serverTime: Date.now() });
}

export async function fingerprint(): Promise<string> {
  const raw = [
    navigator.userAgent,
    navigator.language,
    (navigator as any).hardwareConcurrency,
    screen.width,
    screen.height,
    screen.colorDepth,
    Intl.DateTimeFormat().resolvedOptions().timeZone,
  ].join("|");
  const buf = await crypto.subtle.digest(
    "SHA-256",
    new TextEncoder().encode(raw),
  );
  return Array.from(new Uint8Array(buf))
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("")
    .slice(0, 32);
}

"use client";
import { createContext, useContext, useEffect, useState } from "react";

import { getSetting, setSetting } from "./db";
import { en } from "./i18n/en";
import { ur } from "./i18n/ur";

const dicts = { en, ur } as const;
type Lang = keyof typeof dicts;
const Ctx = createContext<{
  lang: Lang;
  t: (k: string) => string;
  setLang: (l: Lang) => void;
}>({ lang: "en", t: (k) => k, setLang: () => {} });

export function LanguageProvider({ children }: { children: React.ReactNode }) {
  const [lang, setLangState] = useState<Lang>("en");
  useEffect(() => {
    getSetting("lang").then((l) => {
      if (l === "ur" || l === "en") apply(l);
    });
  }, []);
  const apply = (l: Lang) => {
    setLangState(l);
    document.documentElement.lang = l;
    document.documentElement.dir = l === "ur" ? "rtl" : "ltr";
  };
  const setLang = (l: Lang) => {
    apply(l);
    setSetting("lang", l);
  };
  const t = (k: string) => dicts[lang][k] ?? en[k] ?? k;
  return <Ctx.Provider value={{ lang, t, setLang }}>{children}</Ctx.Provider>;
}
export const useT = () => useContext(Ctx);

import nacl from "tweetnacl";
import { getSetting, setSetting } from "./db";

export const WARNING_DAYS = 7;
export const GRACE_DAYS = 3; // read-only days after expiry; set 0 for a hard stop on expiry day
const PUB = process.env.NEXT_PUBLIC_LICENSE_PUBLIC_KEY!;

export interface LicensePayload {
  cid: string;
  fp: string;
  ds: string;
  iat: number;
  exp: string;
  plan: string;
  status: "active" | "revoked";
  features: Record<string, boolean>;
  branding: Record<string, string | number>;
}
export type LicState =
  | "none"
  | "active"
  | "warning"
  | "grace"
  | "locked"
  | "tampered";

const utf8 = new TextEncoder();
const hexToBytes = (h: string) =>
  new Uint8Array(h.match(/../g)!.map((x) => parseInt(x, 16)));
const b64 = {
  enc: (s: string) => btoa(unescape(encodeURIComponent(s))),
  dec: (s: string) => decodeURIComponent(escape(atob(s))),
};

/** 256-bit secret generated once per laptop — its identity, and sync/auth credential */
export async function deviceSecret(): Promise<string> {
  let ds = localStorage.getItem("_ds_v1") ?? (await getSetting("ds"));
  if (!ds) {
    ds = b64.enc(String.fromCharCode(...nacl.randomBytes(32)));
    localStorage.setItem("_ds_v1", ds);
    await setSetting("ds", ds);
  }
  return ds;
}

function verifySig(d: LicensePayload, s: string): boolean {
  try {
    return nacl.sign.detached.verify(
      utf8.encode(JSON.stringify(d)),
      hexToBytes(s),
      hexToBytes(PUB),
    );
  } catch {
    return false;
  }
}

export async function activateWithCode(
  code: string,
  myDs: string,
): Promise<LicensePayload> {
  const { d, s } = JSON.parse(b64.dec(code.trim()));
  if (!verifySig(d, s)) throw new Error("INVALID_CODE");
  if (d.status !== "active") throw new Error("REVOKED");
  if (d.ds !== myDs) throw new Error("DEVICE_MISMATCH");
  const now = String(Date.now());
  await Promise.all([
    setSetting("license", { d, s }),
    setSetting("licenseB", { d, s }),
    setSetting("lastSeen", Date.now()),
    new Promise<void>((r) => {
      localStorage.setItem("_ls_v1", now);
      r();
    }),
  ]);
  return d;
}

export async function getLicense(): Promise<LicensePayload | null> {
  return (
    ((await getSetting("license")) ?? (await getSetting("licenseB")))?.d ?? null
  );
}
export async function getBranding(): Promise<Record<string, any>> {
  return (await getLicense())?.branding ?? {};
}

/** Runs on every launch, fully offline. Re-verifies the signature each time. */
export async function evaluateLicense(): Promise<{
  state: LicState;
  daysLeft: number;
  lic: LicensePayload | null;
}> {
  const stored =
    (await getSetting("license")) ?? (await getSetting("licenseB"));
  const lastSeen = Math.max(
    Number(localStorage.getItem("_ls_v1")) || 0,
    ((await getSetting("lastSeen")) as number) || 0,
  );
  const touch = () =>
    Promise.all([
      setSetting("lastSeen", Date.now()),
      new Promise<void>((r) => {
        localStorage.setItem("_ls_v1", String(Date.now()));
        r();
      }),
    ]);

  if (Date.now() < lastSeen - 86_400_000)
    return { state: "tampered", daysLeft: 0, lic: null }; // clock rolled back
  await touch();
  if (!stored) return { state: "none", daysLeft: 0, lic: null };
  if (!verifySig(stored.d, stored.s))
    return { state: "tampered", daysLeft: 0, lic: null }; // expiry edited locally

  const lic = stored.d;
  const daysLeft = Math.ceil(
    (new Date(lic.exp + "T23:59:59").getTime() - Date.now()) / 86_400_000,
  );
  const state =
    daysLeft > WARNING_DAYS
      ? "active"
      : daysLeft > 0
        ? "warning"
        : daysLeft > -GRACE_DAYS
          ? "grace"
          : "locked";
  return { state, daysLeft, lic };
}

export const canSell = (s: LicState) => s === "active" || s === "warning";
export const canBrowse = (s: LicState) =>
  ["active", "warning", "grace"].includes(s);
export const clockIsSane = (serverTime: number) =>
  Math.abs(Date.now() - serverTime) < 48 * 3_600_000;

const H = () => ({
  Authorization: `Bearer ${process.env.UPSTASH_REDIS_REST_TOKEN}`,
});
const B = () => process.env.UPSTASH_REDIS_REST_URL!;

export async function kvGet<T = any>(key: string): Promise<T | null> {
  const j = await (
    await fetch(`${B()}/get/${encodeURIComponent(key)}`, {
      headers: H(),
      cache: "no-store",
    })
  ).json();
  return j.result ? JSON.parse(j.result) : null;
}
export async function kvSet(key: string, value: unknown) {
  await fetch(`${B()}/set/${encodeURIComponent(key)}`, {
    method: "POST",
    headers: H(),
    body: JSON.stringify(value),
  });
}
export async function kvIncr(key: string): Promise<number> {
  const j = await (
    await fetch(`${B()}/incr/${encodeURIComponent(key)}`, {
      headers: H(),
      cache: "no-store",
    })
  ).json();
  const n = Number(j.result);
  if (n === 1)
    await fetch(`${B()}/expire/${encodeURIComponent(key)}/3600`, {
      headers: H(),
    });
  return n;
}

import { NextRequest, NextResponse } from "next/server";
export function proxy(req: NextRequest) {
  if (
    req.nextUrl.pathname.startsWith("/admin") &&
    req.nextUrl.pathname !== "/admin/login" &&
    !req.cookies.get("admin_sess")
  )
    return NextResponse.redirect(new URL("/admin/login", req.url));
  return NextResponse.next();
}
export const config = { matcher: ["/admin/:path*"] };
